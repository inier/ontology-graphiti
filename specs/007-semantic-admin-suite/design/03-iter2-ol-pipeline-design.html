③a. Candidate 存储 · SQLite DDL

-- 表 1: OL 运行批次（一次文档上传 = 一个 pipeline_run）
CREATE TABLE IF NOT EXISTS usl_pipeline_runs (
  id            TEXT PRIMARY KEY,           -- uuid
  domain_id     TEXT,                       -- 关联的 usl_domains.id（可选，用户未选则为 NULL）
  workspace_id  TEXT NOT NULL,              -- ODAP 工作空间隔离
  doc_sources_json TEXT NOT NULL,           -- JSON: [{source, title, url, checksum}]
  status        TEXT NOT NULL,              -- DRAFT / RUNNING / L1_DONE / L2_DONE / L3_DONE / ...
  started_at    TEXT NOT NULL,
  finished_at   TEXT,
  error_message TEXT,
  stats_json    TEXT NOT NULL DEFAULT '{}'  -- 每一层候选数统计 JSON
);

-- 表 2: Schema Candidate（核心）
CREATE TABLE IF NOT EXISTS usl_schema_candidates (
  id                     TEXT PRIMARY KEY,
  run_id                 TEXT NOT NULL REFERENCES usl_pipeline_runs(id) ON DELETE CASCADE,
  canonical              TEXT NOT NULL,
  en                     TEXT NOT NULL DEFAULT '',
  semantic_type          TEXT NOT NULL,
  synonyms_json          TEXT NOT NULL DEFAULT '[]',
  aliases_json           TEXT NOT NULL DEFAULT '[]',
  origin                 TEXT NOT NULL,   -- usl | llm | hybrid(=混合) | human
  cluster_confidence     REAL NOT NULL DEFAULT 0,
  usl_align_confidence   REAL NOT NULL DEFAULT 0,
  review_confidence      REAL NOT NULL DEFAULT 0,   -- 留空，Iter 3 审批后写入
  status                 TEXT NOT NULL DEFAULT 'DRAFT',
  -- 状态机: DRAFT → L2_DONE → PENDING_REVIEW(3审) → APPROVED / MODIFIED / REJECTED → WRITTEN_BACK(4)
  domain_id              TEXT,                       -- 最终推荐归属的 USL 域
  doc_refs_json          TEXT NOT NULL DEFAULT '[]', -- 文档证据
  parent_candidates_json TEXT NOT NULL DEFAULT '[]', -- L3 分类层级草稿
  created_at             TEXT NOT NULL,
  updated_at             TEXT NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_cand_run ON usl_schema_candidates(run_id);
CREATE INDEX IF NOT EXISTS idx_cand_status ON usl_schema_candidates(status);
CREATE INDEX IF NOT EXISTS idx_cand_sem_type ON usl_schema_candidates(semantic_type);

-- 表 3: 每一层的原始输入/输出快照（调试用，保留 30 天）
CREATE TABLE IF NOT EXISTS usl_pipeline_layer_snapshots (
  id           TEXT PRIMARY KEY,
  run_id       TEXT NOT NULL REFERENCES usl_pipeline_runs(id) ON DELETE CASCADE,
  layer_name   TEXT NOT NULL,  -- L1/L2/L3/L4/L5/L6
  input_json   TEXT NOT NULL,
  output_json  TEXT NOT NULL,
  duration_ms  INTEGER NOT NULL,
  created_at   TEXT NOT NULL
);

🔐 命名空间隔离：所有 Candidate 节点标签前缀 :USL__，避免污染 ODAP 生产 Ontology TBox/ABox。

③c. 双写一致性（弱一致，可读即可）

SQLite 为主，Neo4j 为从（可降级）。
✅ SQLite 写成功但 Neo4j 失败 → 记录 error_message + console warning，不阻断；下一次 /pipeline retry 会 idempotent 覆盖。
❌ SQLite 失败 → 直接抛异常，状态回滚到 RUNNING → ERROR。

④ 后端 API 路由：/api/semantic-admin/pipeline + /candidates（共 7 接口）

Method路径功能关键参数 / 返回结构
POST/pipeline/runs新建运行 → 立即开始 L1→L2 异步任务（asyncio.create_task）Body: {workspace_id, domain_id?, doc_sources:[]}
返回: {run_id, status:'RUNNING'}
GET/pipeline/runs分页列出流水线历史（按工作空间过滤）返回: {items:[{run_id, doc_count, created_at, status, progress:{L1:100,L2:87,...}}]}
GET/pipeline/runs/{id}查看单运行详情（含每层 stats）返回含: {stats:{L1_raw:234, L1_filtered:147, L2_clusters:62}, snapshots_preview: '前 10 条...'}
POST/pipeline/runs/{id}/retry幂等重试失败层（或 force 重跑全部）Query: ?from_layer=L1|L2&force=true/false
GET/candidates按条件搜索候选（run_id · semantic_type · origin · min_confidence · status）返回: {items:SchemaCandidate[], total, page, page_size}
GET/candidates/{id}单个候选详情（含 doc_refs 展开、同义词、置信度）返回全量 JSON
GET/candidates/{id}/graph-preview从 Neo4j _schema_graph 返回该候选的局部可视化（前后端传 Cytoscape JSON）返回 Cytoscape 格式: {nodes[], edges[]}
