// 图谱生成组件
const handleGenerate = async (config: GraphConfig) => {
  const { task_id } = await api.createGraphTask(config);
  // 轮询进度
  const interval = setInterval(async () => {
    const { progress, status } = await api.getProgress(task_id);
    setProgress(progress);
    if (status === 'completed') {
      clearInterval(interval);
      message.success('生成完成');
    }
  }, 1000);
};
